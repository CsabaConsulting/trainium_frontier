56:const std::string* CompilationCacheKey::GetPersistentKey() const {
57:  if (base_key_.empty()) {
58:    return nullptr;
59:  }
60:  // When HLO-content keying is enabled, the persistent key requires HLO
61:  // bytes — without them, persistent caching is not possible (e.g.,
62:  // NeffDirectKernelExecution paths). Caller must skip the persistent cache
63:  // lookup when this returns nullptr.
64:  // When disabled (legacy / production default), persistent key derives from
65:  // base_key, which is always present once we pass the IsEmpty check above.
66:  if (utils::IsHloContentKeyEnabled() && (hlo_bytes_ptr_ == nullptr || hlo_bytes_ptr_->empty())) {
67:    return nullptr;
68:  }
69:  if (persistent_key_.empty()) {
70:    persistent_key_ = ComputePersistentKey();
71:    key_str_.clear();
72:  }
73:  return &persistent_key_;
74:}

76:std::string CompilationCacheKey::ComputePersistentKey() const {
77:  // Persistent NEFF key composition is gated by TORCH_NEURONX_NEFF_KEY_FROM_HLO:
78:  //   - on  (dev/CI): hash(hlo_bytes || compiler_config) — content-addressed.
79:  //                   Lets benign Python source edits re-trace/re-lower (cheap)
80:  //                   but still hit the on-disk NEFF cache when HLO bytes are
81:  //                   unchanged, enabling cross-commit cache reuse.
82:  //   - off (default/prod): hash(base_key || compiler_config) — legacy.
83:  //                   Any source change in base_key invalidates the NEFF.
84:  // GetPersistentKey() guarantees hlo_bytes_ptr_ is valid when the flag is on.
85:  std::string compiler_config = BuildCompilerExtension();
86:  const bool use_hlo = utils::IsHloContentKeyEnabled();
87:  const size_t input_size = use_hlo ? hlo_bytes_ptr_->size() : base_key_.size();
88:
89:  std::vector<uint8_t> combined;
90:  combined.reserve(input_size + 1 + compiler_config.size());
91:
92:  if (use_hlo) {
93:    combined.insert(combined.end(), hlo_bytes_ptr_->begin(), hlo_bytes_ptr_->end());
94:  } else {
95:    combined.insert(combined.end(), reinterpret_cast<const uint8_t*>(base_key_.data()),
96:                    reinterpret_cast<const uint8_t*>(base_key_.data()) + base_key_.size());
97:  }
98:  combined.push_back(kFieldSeparator);
99:
100:  combined.insert(
101:      combined.end(), reinterpret_cast<const uint8_t*>(compiler_config.data()),
102:      reinterpret_cast<const uint8_t*>(compiler_config.data()) + compiler_config.size());
103:
104:  // 32-char hex string
105:  return cache_utils::ComputeXXH3Hash128(combined);
106:}

19:/**
20: * @brief Unified cache key for compiled NEFFs.
21: *
22: * The key supports two levels optimized for different use cases:
23: *
24: * 1. In-memory key: `base_key`
25: *    Used for in-memory cache lookup. The `base_key` is a string from the
26: *    frontend (e.g., op_name + shapes + dtypes from eager, or a hash string
27: *    from torch.compile). Includes Python source/version info so that any
28: *    upstream change forces a re-trace/re-lower.
29: *
30: * 2. Persistent key: gated by TORCH_NEURONX_NEFF_KEY_FROM_HLO.
31: *    Computed lazily when persistent cache access is needed.
32: *
33: *    - Flag on  (dev/CI):  hash(hlo_bytes || compiler_config)
34: *      Content-derived from the lowered HLO/MLIR bytes plus compiler config
35: *      (platform, lnc, optimization level, compiler args, neuronx-cc version).
36: *      Does NOT include base_key — benign Python edits re-trace/re-lower
37: *      (cheap) but still hit the on-disk NEFF cache when HLO bytes are
38: *      unchanged, enabling cross-commit cache reuse.
39: *      Returns nullptr if HLO bytes are not available (e.g., NeffDirect
40: *      paths); callers gracefully skip the persistent cache in that case.
41: *
42: *    - Flag off (default/prod):  hash(base_key || compiler_config)
43: *      Legacy behavior. Any change in base_key (source/version_hash) busts
44: *      the persistent cache. Always available once the key is non-empty.
45: *
46: * ## In-memory key identity:  base_key
47: * ## Persistent key identity (flag on):  hash(hlo_bytes || compiler_config)
48: * ## Persistent key identity (flag off): hash(base_key || compiler_config)
49: *
50: */
