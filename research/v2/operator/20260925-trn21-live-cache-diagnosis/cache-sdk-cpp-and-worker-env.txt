108:void PersistentCacheBackend::AtomicWrite(const std::filesystem::path& path,
109:                                         const std::vector<uint8_t>& data) {
110:  // Create parent directories
111:  std::error_code ec;
112:  std::filesystem::create_directories(path.parent_path(), ec);
113:  if (ec) {
114:    throw std::runtime_error("Failed to create directory: " + path.parent_path().string() + " (" +
115:                             ec.message() + ")");
116:  }
117:
118:  // Create temporary file atomically using mkstemp (POSIX)
119:  // mkstemp creates a unique file and returns an open fd - race-condition free
120:  std::string temp_template = (path.parent_path() / ".tmp.XXXXXX").string();
121:  std::vector<char> temp_path_buf(temp_template.begin(), temp_template.end());
122:  temp_path_buf.push_back('\0');
123:
124:  int fd = mkstemp(temp_path_buf.data());
125:  if (fd == -1) {
126:    throw std::runtime_error("Failed to create temporary file: " + std::string(strerror(errno)));
127:  }
128:
129:  std::filesystem::path temp_path(temp_path_buf.data());
130:
131:  // Write data directly using POSIX write
132:  const uint8_t* write_ptr = data.data();
133:  size_t remaining = data.size();
134:  while (remaining > 0) {
135:    ssize_t written = write(fd, write_ptr, remaining);
136:    if (written == -1) {
137:      int err = errno;
138:      close(fd);
139:      std::filesystem::remove(temp_path, ec);
140:      throw std::runtime_error("Failed to write to temporary file: " + std::string(strerror(err)));
141:    }
142:    write_ptr += written;
143:    remaining -= static_cast<size_t>(written);
144:  }
145:
146:  if (close(fd) == -1) {
147:    int err = errno;
148:    std::filesystem::remove(temp_path, ec);
149:    throw std::runtime_error("Failed to close temporary file: " + std::string(strerror(err)));
150:  }
151:
152:  // Atomic rename
153:  std::filesystem::rename(temp_path, path, ec);
154:  if (ec) {
155:    // Clean up temp file on rename failure
156:    std::filesystem::remove(temp_path);
157:    throw std::runtime_error("Failed to rename temporary file: " + temp_path.string() + " -> " +
158:                             path.string() + " (" + ec.message() + ")");
159:  }

190:bool CompilationCache::CheckInMemoryCacheAndPending(
191:    const CompilationCacheKey& key, const CompilableKernelExecution& kernel_execution) {
192:  const std::string& cache_key = key.GetKey();
193:  // Check for completed NEFF in in-memory cache.
194:  auto cache_it = cache_.find(cache_key);
195:  if (cache_it != cache_.end()) {
196:    auto cached_entry = cache_it->second;
197:    TORCH_CHECK(cached_entry != nullptr, "Cache entry can not be null");
198:
199:    // Fast-fail if this entry records a previous compilation failure.
200:    if (cached_entry->IsFailed()) {
201:      ThrowCompilationError(key, cached_entry->GetFailureMessage());
202:    }
203:
204:    // Record in-memory cache hit and increment access statistics.
205:    TORCH_NEURONX_COUNTER("CompilationCache.InMemoryHits", 1);
206:    cached_entry->UpdateAccess();
207:
208:    // Create managed pointer and set on kernel execution.
209:    auto neff_ptr = CreateNeffBytesPtr(cached_entry->neff_bytes, cache_key);
210:    kernel_execution.SetCachedNeff(std::move(neff_ptr));
211:    TORCH_NEURONX_DEBUG("In-memory cache hit: ", key.ToString());
212:    return true;
213:  }
214:
215:  // No completed NEFF found - check if compilation is already pending.
216:  auto pending_it = pending_compilations_.find(cache_key);
217:  if (pending_it != pending_compilations_.end()) {
218:    // Another thread is compiling, count as in-memory hit since we'll get the result.
219:    TORCH_NEURONX_COUNTER("CompilationCache.InMemoryHits", 1);
220:    return true;
221:  }
222:
223:  return false;
224:}
